  1. CONCEPT GÉNÉRAL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Des "coders" sont assis en cercle autour d'une table. Entre chaque paire de
  coders voisins se trouve une clé USB (dongle). Pour compiler, un coder a besoin
  de ses 2 dongles voisins EN MÊME TEMPS. Après compilation, il débogue, refactorise,
  puis recommence. S'il n'arrive pas à compiler avant le délai (time_to_burnout),
  il "brûle" (burnout) et la simulation s'arrête.

  Cercle de 5 coders (exemple) :

       Coder 1
      /       \
  dongle5   dongle1
    /             \
 Coder 5       Coder 2
    \             /
  dongle4   dongle2
      \       /
       Coder 4 — dongle3 — Coder 3

  → Coder 1 a besoin de dongle5 ET dongle1 simultanément.
  → Coder 1 et Coder 3 peuvent compiler EN MÊME TEMPS (aucun dongle partagé).

  │    int *wait_queue  → tableau des IDs dans la heap      │

  │                        coder i+1 (0 = pas dans la queue)│

  ORDRE DES VERROUS (toujours respecté pour éviter deadlock) :
  ┌──────────────┐    puis    ┌──────────────┐
  │  print_lock  │ ─────────► │  sched_lock  │
  └──────────────┘            └──────────────┘
  On prend toujours print_lock AVANT sched_lock.
  Jamais l'inverse.

  5. VALIDATION DES ARGUMENTS  [src/validate.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  validate_args(argc, argv)
  ├── argc != 9 → affiche l'usage, retourne 0
  ├── Pour chaque argv[1..7] :
  │     is_number()  → doit être un entier non négatif
  │     argv[1..6]   → doit être > 0 (pas zéro, sauf argv[6]=nb_compiles_req)
  │     argv[7]      → dongle_cooldown : >= 0 autorisé (peut être 0)
  └── argv[8] → doit être exactement "fifo" ou "edf"

  Règle spéciale :
    argv[6] = nb_compiles_req peut être 0 (= tourner jusqu'au burnout)
    argv[7] = dongle_cooldown peut être 0 (= pas de cooldown)
    Tous les autres arguments doivent être > 0.

  6. INITIALISATION  [src/init.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  init_system(data)
  ├── allocate_system()
  │     ├── malloc(nb_coders * sizeof(t_dongle))  → tableau de N dongles
  │     ├── malloc(nb_coders * sizeof(t_coder))   → tableau de N coders
  │     ├── calloc(nb_coders, sizeof(int))         → wait_queue (zéro-initialisé)
  │     └── calloc(nb_coders, sizeof(int))         → wait_order (zéro-initialisé)
  │
  ├── init_sync_objects()
  │     ├── pthread_mutex_init(&print_lock)
  │     ├── pthread_mutex_init(&sched_lock)
  │     └── pthread_cond_init(&sched_cond)
  │
  ├── init_coders()  [boucle sur i=0..N-1]
  │     └── init_dongle_and_coder(data, i)
  │           ├── dongles[i].is_used = 0
  │           ├── dongles[i].available_at = 0
  │           ├── coders[i].id = i + 1          ← IDs commencent à 1
  │           ├── coders[i].nb_compiles = 0
  │           ├── coders[i].left_dongle  = &dongles[i]
  │           ├── coders[i].right_dongle = &dongles[(i+1) % N]
  │           │     ← Le dernier coder (N) partage avec le 1er : dongle circulaire
  │           └── coders[i].data = data
  │
  └── data->wait_count = 0
      data->next_order = 1
      data->sim_running = 1

  7. LANCEMENT DES THREADS  [src/threads.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  init_threads(data, monitor)
  │
  ├── data->start_time = current_time_ms()   ← horodatage T=0
  │
  ├── LOCK sched_lock
  │   ├── Boucle : met à jour last_compile_start pour chaque coder
  │   └── enqueue_all(data)                  ← PRÉ-ENREGISTREMENT (important !)
  │         ├── Enregistre d'abord les coders IMPAIRS : 1, 3, 5, ...
  │         │     → ils obtiennent les ordres FIFO 1, 2, 3, ... (priorité max)
  │         └── Puis les coders PAIRS : 2, 4, 6, ...
  │               → ils obtiennent les ordres FIFO suivants
  ├── UNLOCK sched_lock
  │
  ├── Crée les threads coders 0..N-1 → chacun exécute coder_routine()
  └── Crée le thread monitor         → exécute monitor_thread()

  POURQUOI PRÉ-ENREGISTRER ?
  Sans pré-enregistrement, c'est le scheduler OS qui décide quel thread
  s'enregistre en premier → ordre FIFO aléatoire → coders pairs peuvent
  passer avant des coders impairs voisins (bugué). En pré-enregistrant
  impairs puis pairs, le coder 1 a toujours la priorité 1, le coder 3
  la priorité 2, etc. → comportement déterministe correct.

  8. ROUTINE D'UN CODER  [src/routine.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  coder_routine(arg)     ← chaque coder est un thread qui exécute ça
  │
  └── BOUCLE PRINCIPALE (tant que simulation active) :
        │
        ├── 1. scheduler_request(data, coder)
        │         → "Je veux compiler, je m'inscris dans la file d'attente
        │            et j'attends que mes 2 dongles soient disponibles"
        │         → retourne 0 si simulation arrêtée → sort de la boucle
        │
        ├── 2. start_compile() → print "X is compiling"
        │
        ├── 3. sleep_ms(time_to_compile)
        │         → dort pendant la durée de compilation
        │         → implémenté avec usleep(500) en boucle pour rester réactif
        │
        ├── 4. count_compile()
        │         → LOCK sched_lock
        │         → coder->nb_compiles++
        │         → UNLOCK sched_lock
        │
        ├── 5. scheduler_release(data, coder)
        │         → relâche les 2 dongles avec cooldown
        │         → available_at = now + dongle_cooldown
        │         → broadcast sur sched_cond pour réveiller les waiters
        │
        ├── 6. print "X is debugging"  → sleep_ms(time_to_debug)
        │
        └── 7. print "X is refactoring" → sleep_ms(time_to_refactor)
              → retour au début de la boucle

  9. ORDONNANCEUR  [src/scheduler.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  scheduler_request(data, coder)
  ├── LOCK sched_lock
  ├── sched_enqueue(data, my_id)   ← s'inscrit dans la file de priorité
  │                                   (si déjà inscrit = no-op)
  │
  └── BOUCLE INFINIE :
        ├── si !sim_running → dequeue, UNLOCK, return 0  (arrêt simulation)
        ├── try_acquire(data, coder, my_id)
        │     → si les conditions sont réunies, PREND les dongles
        │     → retourne 1 : sort de la boucle
        └── sinon wait_scheduler(data)
              → pthread_cond_wait(&sched_cond, &sched_lock)
              → se rendort, sera réveillé par un broadcast

  ─────────────────────────────────────────────────────────
  try_acquire(data, coder, my_id)     ← ACQUISITION ATOMIQUE
  │
  ├── coder_can_compile(coder) ?
  │     → les 2 dongles sont libres ET disponibles (cooldown expiré) ?
  │
  ├── priority_is_clear(data, my_id) ?
  │     → personne de priorité supérieure partageant un de mes dongles
  │       ne peut compiler en ce moment ?
  │
  └── SI OUI aux deux :
        ├── left_dongle->is_used  = 1   ← ATOMIQUEMENT sous sched_lock
        ├── right_dongle->is_used = 1
        ├── last_compile_start = now     ← remet le chrono burnout à zéro
        ├── sched_dequeue(data, my_id)   ← quitte la file
        ├── broadcast(sched_cond)        ← réveille les autres waiters
        ├── UNLOCK sched_lock
        ├── print_dongle_taken()         ← affiche "X has taken a dongle" (×2)
        ├── LOCK sched_lock
        └── return 1

  ─────────────────────────────────────────────────────────
  scheduler_release(data, coder)
  ├── when = now + dongle_cooldown
  ├── LOCK sched_lock
  ├── left_dongle->is_used    = 0
  ├── left_dongle->available_at = when   ← période de refroidissement
  ├── right_dongle->is_used   = 0
  ├── right_dongle->available_at = when
  ├── broadcast(sched_cond)              ← réveille tous les waiters
  └── UNLOCK sched_lock

  10. FILE DE PRIORITÉ  [src/scheduler_queue.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  STRUCTURE : Min-Heap (tas-min)
  ─────────────────────────────
  Un tableau wait_queue[] organisé comme un arbre binaire.
  La "plus petite" valeur (la plus haute priorité) est toujours à l'index 0.

  Exemple avec 4 coders, FIFO, tous inscrits :

       wait_queue = [1, 2, 3, 4]    ← coder 1 au sommet (priorité 1)
                         1
                        / \
                       2   3
                      /
                     4

  ─────────────────────────────────────────────────────────
  priority_before(data, first_id, second_id)
  → "first_id est-il plus prioritaire que second_id ?"

  MODE FIFO :
    return (wait_order[first_id-1] < wait_order[second_id-1])
    → celui qui s'est inscrit en PREMIER est prioritaire

  MODE EDF (Earliest Deadline First) :
    first_deadline  = last_compile_start[first_id]  + time_to_burnout
    second_deadline = last_compile_start[second_id] + time_to_burnout
    si deadlines égaux → return (first_id < second_id)  ← ID plus bas gagne
    sinon → return (first_deadline < second_deadline)
    → celui dont le burnout est le PLUS PROCHE est prioritaire

  ─────────────────────────────────────────────────────────
  sched_enqueue(data, coder_id)
  ├── si déjà inscrit (wait_order != 0) → return (no-op)
  ├── si file pleine (wait_count >= N)  → return
  ├── wait_order[coder_id-1] = next_order++   ← attribue un numéro d'ordre
  ├── wait_queue[wait_count] = coder_id        ← ajoute en fin de tableau
  ├── heap_up(wait_count)                      ← remonte pour maintenir l'ordre
  └── wait_count++

  sched_dequeue(data, coder_id)
  ├── trouve l'index i de coder_id dans wait_queue[]
  ├── wait_order[coder_id-1] = 0              ← marque comme sorti
  ├── wait_queue[i] = wait_queue[--wait_count] ← remplace par le dernier
  └── heap_up() sur tout le tableau           ← reconstruit la propriété du tas

  heap_up(data, index)
  → compare l'élément à son parent (index-1)/2
  → si plus prioritaire, échange et remonte
  → répète jusqu'à la racine ou ordre correct

  11. VÉRIFICATIONS DU SCHEDULER  [src/scheduler_checks.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  coder_can_compile(coder)
  → retourne 1 si ET SEULEMENT SI :
    - left_dongle->is_used  == 0   ← dongle gauche libre
    - right_dongle->is_used == 0   ← dongle droit libre
    - now >= left_dongle->available_at   ← cooldown gauche expiré
    - now >= right_dongle->available_at  ← cooldown droit expiré

  ─────────────────────────────────────────────────────────
  shares_dongle(data, first_id, second_id)
  → deux coders partagent-ils un dongle ?
    first_left  = first_id - 1
    first_right = first_id % N
    second_left  = second_id - 1
    second_right = second_id % N
  → retourne 1 si une des 4 combinaisons (left/right × left/right) est identique

  Exemple (N=5) : coder 2 et coder 4
    coder2 : left=1, right=2   |  coder4 : left=3, right=4
    → aucune valeur commune → shares_dongle = 0 → pas de conflit

  Exemple (N=5) : coder 4 et coder 5
    coder4 : left=3, right=4   |  coder5 : left=4, right=0
    → right(4)==left(5)==4 → shares_dongle = 1 → CONFLIT

  ─────────────────────────────────────────────────────────
  priority_is_clear(data, my_id)
  → "Puis-je passer sans voler la priorité à quelqu'un ?"

  Parcourt TOUS les coders dans wait_queue :
    Pour chaque "other_id" :
      SI  shares_dongle(my_id, other_id)           ← on partage un dongle
      ET  priority_before(other_id, my_id)         ← il a plus haute priorité
      ET  coder_can_compile(&coders[other_id-1])   ← il PEUT compiler maintenant
      → ALORS je dois attendre → return 0

  Règle finale : return (wait_order[my_id-1] != 0)
    → si je ne suis même pas dans la file, retourne 0

  13. THREAD MONITEUR  [src/monitor.c]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  monitor_thread(arg)
  └── BOUCLE : while (monitor_cycle(data))
                   usleep(100)   ← vérifie toutes les 100 µs
                                    (aussi = source de broadcast régulier)

  monitor_cycle(data)
  ├── LOCK sched_lock
  ├── si !sim_running → UNLOCK, return 0    ← simulation déjà arrêtée
  │
  ├── check_all_burnouts(data)
  │     → Pour chaque coder i :
  │           check_burnout(data, &coders[i])
  │           ├── si nb_compiles >= nb_compiles_req (déjà fini) → skip
  │           ├── now - last_compile_start > time_to_burnout ?
  │           │     OUI → sim_running = 0
  │           │         → broadcast (pour réveiller tout le monde)
  │           │         → UNLOCK sched_lock
  │           │         → print "X burned out" (rouge)
  │           │         → return 1
  │           └── NON → return 0
  │
  ├── all_done(data) ?
  │     → si nb_compiles_req == 0 → return 0 (jamais "done")
  │     → si TOUS les coders ont nb_compiles >= nb_compiles_req → return 1
  │
  ├── si all_done : sim_running = 0
  ├── broadcast(sched_cond)
  ├── UNLOCK sched_lock
  │
  └── si all_done :
        LOCK print_lock
        printf "All coders have compiled." (vert)
        UNLOCK print_lock

  POURQUOI LE MONITEUR BROADCAST ?
  Le moniteur fait un broadcast toutes les 100µs. Cela sert de "réveil
  périodique" pour les coders en attente. Même sans timedwait, les coders
  sont réveillés régulièrement pour vérifier si leurs dongles sont disponibles
  (important pour le cooldown des dongles).

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    LOCK print_lock
    ts = current_time_ms() - start_time   ← timestamp depuis T=0
    si (simulation active OU message de burnout) → printf
    UNLOCK print_lock

  print_state(data, id, msg)
  → Affichage générique : "123 4 is compiling"
  → Si msg[0] == 'b' (burned out) → affiché en ROUGE avec saut de ligne
    + s'affiche même si sim_running == 0 (burnout toujours visible)

  print_dongle_taken(data, id)
  → Affiche "123 4 has taken a dongle" DEUX FOIS avec le MÊME timestamp
  → Les deux lignes sont dans le MÊME lock → même milliseconde garantie
  → Les 2 acquisitions sont atomiques donc les 2 prints doivent être identiques

  print_compile(data, id)
  → Affiche "123 4 is compiling"

  stop_simulation(data)
  → LOCK sched_lock → sim_running = 0 → broadcast → UNLOCK

  simulation_is_running(data)
  → LOCK sched_lock → lit sim_running → UNLOCK → retourne la valeur
  → Utilisé dans print_state pour ne pas afficher si simulation terminée

  18. CONDITIONS D'ARRÊT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  La simulation s'arrête dans deux cas :

  CAS 1 — BURNOUT :
    Le moniteur détecte :
      now - last_compile_start > time_to_burnout
    → sim_running = 0
    → broadcast (tous les coders se réveillent)
    → chaque coder voit !sim_running dans scheduler_request → return 0
    → les threads se terminent proprement

  CAS 2 — TOUS ONT COMPILÉ :
    Le moniteur détecte que nb_compiles_req > 0 ET
    tous les coders ont nb_compiles >= nb_compiles_req
    → sim_running = 0
    → broadcast
    → affiche "All coders have compiled." en vert

  CAS 3 — nb_compiles_req = 0 :
    La simulation tourne INDÉFINIMENT jusqu'à ce qu'un coder brûle.
    all_done() retourne toujours 0 dans ce cas.

  19. PRÉVENTION DU DEADLOCK — CONDITIONS DE COFFMAN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Un deadlock dans le problème des philosophes classique survient quand
  chaque philosophe tient UN couteau et attend l'autre. Ici, cette situation
  est évitée par ACQUISITION ATOMIQUE.

  Les 4 conditions de Coffman (toutes nécessaires pour un deadlock) :

  1. EXCLUSION MUTUELLE    → OUI (les dongles sont exclusifs)
     [inhérent au problème, ne peut pas être supprimé]

  2. HOLD AND WAIT         → ÉLIMINÉ ✓
     Un coder ne prend JAMAIS un seul dongle en attendant l'autre.
     Il attend dans la file jusqu'à ce que les DEUX soient disponibles,
     puis les prend tous les deux ATOMIQUEMENT sous sched_lock.

  3. PAS DE PRÉEMPTION     → Pas nécessaire
     Puisque le coder n'attend jamais en tenant un dongle, on n'a pas
     besoin de "voler" les dongles.

  4. ATTENTE CIRCULAIRE    → ÉLIMINÉ ✓
     L'arbitrage centralisé (file de priorité) garantit qu'un coder
     de plus haute priorité est servi en premier. Pas de cycle possible.

  21. POLITIQUE DE PRIORITÉ — FIFO vs EDF
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  FIFO (First In First Out)
  ─────────────────────────
  • La priorité = order d'arrivée dans la file (wait_order[])
  • Plus le numéro d'ordre est PETIT → plus haute priorité
  • À la pré-enregistrement : coders impairs (1,3,5...) reçoivent
    les numéros les plus bas → premier round toujours = impairs
  • Empêche la famine : tout le monde est servi dans l'ordre d'arrivée

  EDF (Earliest Deadline First)
  ─────────────────────────────
  • La priorité = urgence du burnout
  • deadline = last_compile_start + time_to_burnout
  • Plus la deadline est PROCHE → plus haute priorité
  • En cas d'égalité → le coder avec l'ID le PLUS BAS gagne
  • Maximise le nombre de coders qui respectent leur deadline
  • Utilisé quand certains coders ont attendu plus longtemps
